function App() {
  const [page, setPage] = useState<Page>('dashboard')
  const [sidebarOpen, setSidebarOpen] = useState(false)
  const [notifications, setNotifications] = useState(false)
  const [profileMenu, setProfileMenu] = useState(false)
  const [toast, setToast] = useState<string | null>(null)
  const [semester, setSemester] = useState('2026.2')
  const [dark, setDark] = useState(() => localStorage.getItem('educonnect-theme') === 'dark')
  const [loggedOut, setLoggedOut] = useState(false)
  const [logoutConfirm, setLogoutConfirm] = useState(false)
  const [toastKind, setToastKind] = useState<'success' | 'error'>('success')
  const notify = (message: string, kind: 'success' | 'error' = 'success') => { setToastKind(kind); showToast(message) }
  const logout = () => setLogoutConfirm(true)
  const toggleTheme = () => { const next = !dark; setDark(next); localStorage.setItem('educonnect-theme', next ? 'dark' : 'light') }


  const navigate = (next: Page) => {
    setPage(next)
    setSidebarOpen(false)
    setProfileMenu(false)
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }

  const showToast = (message: string) => {
    setToast(message)
    window.setTimeout(() => setToast(null), 3000)
  }

  const title = useMemo(() => ({
    dashboard: 'Dashboard',
    notas: 'Minhas notas',
    disciplinas: 'Minhas disciplinas',
    matricula: 'Matrícula',
    perfil: 'Meu perfil',
  }[page]), [page])

  return (
    <div className={`app-root min-h-screen bg-slate-50 ${dark ? 'theme-dark' : ''}`}>
      {loggedOut ? <div className="flex min-h-screen items-center justify-center p-6"><div className="card max-w-md p-8 text-center"><GraduationCap className="mx-auto h-14 w-14 text-indigo-500"/><h1 className="mt-4 text-2xl font-extrabold">Sessão encerrada</h1><p className="mt-2 text-sm text-slate-500">Você saiu do portal de demonstração.</p><button className="btn-primary mt-6" onClick={() => { setLoggedOut(false); navigate('dashboard') }}>Entrar novamente</button></div></div> : <>
      <Sidebar page={page} open={sidebarOpen} onNavigate={navigate} onClose={() => setSidebarOpen(false)} onLogout={logout} />
      <div className="lg:pl-72">
        <Header
          title={title}
          semester={semester}
          setSemester={setSemester}
          onMenu={() => setSidebarOpen(true)}
          dark={dark} onToggleTheme={toggleTheme}
          notifications={notifications}
          setNotifications={setNotifications}
          profileMenu={profileMenu}
          setProfileMenu={setProfileMenu}
          onProfile={() => navigate('perfil')}
          onLogout={logout}
        />
        <main className="mx-auto max-w-[1500px] px-4 pb-28 pt-6 sm:px-6 lg:px-8 lg:pb-10">
          {page === 'dashboard' && <Dashboard onNavigate={navigate} onToast={notify} />}
          {page === 'notas' && <Notas semester={semester} />}
          {page === 'disciplinas' && <Disciplinas />}
          {page === 'matricula' && <Matricula onToast={notify} />}
          {page === 'perfil' && <Perfil onToast={notify} />}
        </main>
      </div>
      <MobileNav page={page} onNavigate={navigate} />
      </>}
      {logoutConfirm && <div className="fixed inset-0 z-[80] grid place-items-center bg-slate-950/60 p-4" role="presentation"><div role="dialog" aria-modal="true" aria-labelledby="logout-title" className="card w-full max-w-sm p-6 shadow-2xl"><h2 id="logout-title" className="text-xl font-bold">Sair do sistema?</h2><p className="mt-2 text-sm text-slate-500">Você poderá entrar novamente na demonstração.</p><div className="mt-6 flex justify-end gap-3"><button className="btn-secondary" onClick={() => setLogoutConfirm(false)}>Cancelar</button><button className="btn-primary" onClick={() => {setLogoutConfirm(false);setLoggedOut(true)}}>Confirmar saída</button></div></div></div>}
      {toast && (
        <div role="status" aria-live="polite" className="fixed bottom-20 left-1/2 z-[90] flex w-max max-w-[calc(100vw-2rem)] -translate-x-1/2 animate-float-in items-center gap-3 rounded-2xl bg-slate-900 px-4 py-3 text-sm font-semibold text-white shadow-2xl">
          {toastKind === 'success' ? <CheckCircle2 className="h-5 w-5 text-emerald-400" /> : <AlertCircle className="h-5 w-5 text-rose-400" />}
          {toast}
        </div>
      )}
    </div>
  )
}

function StatCard({ label, value, detail, icon: Icon, tone }: { label: string; value: string; detail: string; icon: ComponentType<{ className?: string }>; tone: string }) {
  const tones: Record<string, string> = {
    indigo: 'bg-indigo-50 text-indigo-600',
    emerald: 'bg-emerald-50 text-emerald-600',
    violet: 'bg-violet-50 text-violet-600',
    amber: 'bg-amber-50 text-amber-600',
  }
  return <div className="card animate-fade-up p-5 transition duration-300 hover:-translate-y-1 hover:shadow-xl"><div className="flex items-center justify-between"><div className={`grid h-10 w-10 place-items-center rounded-xl ${tones[tone]}`}><Icon className="h-5 w-5" /></div><span className="text-xs font-bold text-slate-400">2026.2</span></div><p className="mt-5 text-sm font-medium text-slate-500">{label}</p><p className="mt-1 text-3xl font-extrabold tracking-tight">{value}</p><p className="mt-1 text-xs font-medium text-slate-400">{detail}</p></div>
}

function Matricula({ onToast }: { onToast: (m: string, kind?: 'success' | 'error') => void }) {
  const [selected, setSelected] = useState<string[]>([])
  const [confirmed, setConfirmed] = useState<string[]>([])
  const [review, setReview] = useState(false)
  const [loading, setLoading] = useState(false)
  const toggle = (code: string) => setSelected(current => current.includes(code) ? current.filter(c => c !== code) : [...current, code])
  const chosen = availableSubjects.filter(s => selected.includes(s.code))
  const credits = chosen.reduce((total, s) => total + s.credits, 24)
  const submit = () => {
    if (!selected.length) return onToast('Selecione pelo menos uma disciplina.', 'error')
    if (credits > 40) return onToast('Limite de 40 créditos excedido.', 'error')
    setReview(true)
  }
  const confirm = async () => {
    setLoading(true)
    await new Promise(resolve => setTimeout(resolve, 1500))
    setConfirmed([...selected]); setReview(false); setLoading(false)
    onToast(`Matrícula confirmada em ${selected.length} disciplina(s)!`)
  }
  return <div>
    <PageHeading eyebrow="Período 2026.2" title="Monte sua grade" description="Selecione as disciplinas, revise sua escolha e confirme a pré-matrícula." action={<span className="rounded-full bg-emerald-50 px-3 py-2 text-xs font-bold text-emerald-700">Matrículas abertas</span>} />
    {confirmed.length > 0 && <div role="status" className="mb-5 rounded-xl border border-emerald-200 bg-emerald-50 p-4 text-sm font-semibold text-emerald-800">Matrícula demonstrativa confirmada: {confirmed.length} disciplina(s).</div>}
    <div className="grid gap-6 xl:grid-cols-[1fr_340px]">
      <section aria-label="Disciplinas disponíveis" className="space-y-3">{availableSubjects.map(s => <label key={s.code} className={`card flex cursor-pointer items-center gap-4 p-4 transition hover:-translate-y-0.5 sm:p-5 ${selected.includes(s.code) ? 'ring-2 ring-indigo-400' : ''}`}><input type="checkbox" checked={selected.includes(s.code)} onChange={() => toggle(s.code)} className="h-5 w-5 shrink-0 accent-indigo-600" /><span className="min-w-0 flex-1"><span className="block font-bold">{s.name}</span><span className="mt-1 block text-xs text-slate-500">{s.code} • {s.professor} • {s.schedule}</span><span className="mt-1 block text-xs text-slate-500">{s.credits} créditos • {s.vacancies} vagas</span></span></label>)}</section>
      <aside className="card h-fit p-5 xl:sticky xl:top-24"><h3 className="text-lg font-extrabold">Resumo da matrícula</h3><p className="mt-1 text-sm text-slate-500">Revise antes de confirmar.</p><div className="my-5 space-y-3 border-y border-slate-100 py-5 text-sm"><div className="flex justify-between"><span>Disciplinas atuais</span><b>6</b></div><div className="flex justify-between"><span>Novas disciplinas</span><b>{selected.length}</b></div><div className="flex justify-between"><span>Total de créditos</span><b>{credits}/40</b></div></div><div className="space-y-2">{chosen.map(s => <div key={s.code} className="flex items-center justify-between rounded-lg bg-slate-50 p-2 text-xs"><span>{s.name}</span><button aria-label={`Remover ${s.name}`} onClick={() => toggle(s.code)}><Trash2 className="h-4 w-4" /></button></div>)}</div><button onClick={submit} disabled={!selected.length} className="btn-primary mt-5 w-full disabled:cursor-not-allowed disabled:opacity-50">Revisar matrícula <ArrowRight className="inline h-4 w-4" /></button></aside>
    </div>
    {review && <div className="fixed inset-0 z-[80] grid place-items-center bg-slate-950/60 p-4"><div role="dialog" aria-modal="true" aria-labelledby="review-title" className="card w-full max-w-lg p-6"><h3 id="review-title" className="text-xl font-extrabold">Confirmar disciplinas</h3><p className="mt-2 text-sm text-slate-500">Confira sua seleção antes de finalizar.</p><ul className="my-5 list-inside list-disc space-y-2 text-sm">{chosen.map(s => <li key={s.code}>{s.name} — {s.credits} créditos</li>)}</ul><p className="font-bold">Total: {credits} créditos</p><div className="mt-6 flex flex-wrap justify-end gap-3"><button disabled={loading} onClick={() => setReview(false)} className="btn-secondary">Voltar</button><button disabled={loading} onClick={confirm} className="btn-primary inline-flex items-center gap-2 disabled:opacity-60">{loading && <LoaderCircle className="h-4 w-4 animate-spin" />}{loading ? 'Confirmando...' : 'Confirmar matrícula'}</button></div></div></div>}
  </div>
}

function Perfil({ onToast }: { onToast: (m: string, kind?: 'success' | 'error') => void }) {
  const initial = { name: 'João Silva', email: 'joao.silva@email.com', phone: '(11) 99999-8888', city: 'São Paulo', course: 'Sistemas de Informação' }
  const [saved, setSaved] = useState(() => { try { return { ...initial, ...JSON.parse(localStorage.getItem('educonnect-profile') || '{}') } } catch { return initial } })
  const [form, setForm] = useState(saved)
  const [loading, setLoading] = useState(false)
  const update = (key: keyof typeof form, value: string) => setForm(f => ({ ...f, [key]: value }))
  const submit = async (e: FormEvent) => {
    e.preventDefault()
    if (form.name.trim().length < 3 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email) || form.phone.replace(/\D/g, '').length < 10 || !form.city.trim()) return onToast('Verifique nome, e-mail, telefone e cidade.', 'error')
    setLoading(true)
    await new Promise(resolve => setTimeout(resolve, 1600))
    localStorage.setItem('educonnect-profile', JSON.stringify(form)); setSaved(form); setLoading(false)
    onToast('Alterações salvas com sucesso!')
  }
  return <div><PageHeading eyebrow="Conta acadêmica" title="Meu perfil" description="Edite seus dados de contato. As alterações são salvas neste navegador para demonstração." />
    <form onSubmit={submit} className="grid gap-6 xl:grid-cols-[300px_1fr]">
      <section className="card h-fit p-6 text-center"><div className="mx-auto grid h-24 w-24 place-items-center rounded-full bg-indigo-600 text-2xl font-extrabold text-white">{form.name.split(' ').map(n => n[0]).slice(0,2).join('').toUpperCase()}</div><h3 className="mt-4 font-extrabold">{form.name}</h3><p className="mt-1 text-xs text-slate-500">RA 20261234</p><div className="mt-5 rounded-xl bg-slate-50 p-3 text-left text-xs"><b>Curso</b><p className="mt-1 text-slate-500">{form.course}</p></div></section>
      <section className="card p-5 sm:p-7"><h3 className="border-b border-slate-100 pb-5 text-lg font-extrabold">Dados pessoais</h3><div className="mt-6 grid gap-5 sm:grid-cols-2"><Field label="Nome completo" icon={User} value={form.name} onChange={v => update('name', v)} required minLength={3} /><Field label="E-mail" icon={Mail} value={form.email} onChange={v => update('email', v)} type="email" required /><Field label="Telefone" icon={Phone} value={form.phone} onChange={v => update('phone', v)} type="tel" required /><Field label="Cidade" icon={MapPin} value={form.city} onChange={v => update('city', v)} required /></div><div className="mt-8 flex flex-wrap justify-end gap-3 border-t border-slate-100 pt-5"><button type="button" disabled={loading} onClick={() => {setForm(saved);onToast('Alterações não salvas descartadas.')}} className="btn-secondary">Restaurar</button><button type="submit" disabled={loading} className="btn-primary inline-flex items-center gap-2 disabled:opacity-60">{loading && <LoaderCircle className="h-4 w-4 animate-spin" />}{loading ? 'Salvando...' : 'Salvar alterações'}</button></div></section>
    </form><section className="card mt-6 p-5 sm:p-6"><h3 className="font-extrabold">Segurança da conta</h3><p className="mt-2 text-sm text-slate-500">Esta é uma demonstração de interface. Troca de senha e sessões reais exigem autenticação e backend.</p></section>
  </div>
}
function Field({ label, icon: Icon, value, onChange, type = 'text', required = false, minLength }: { label: string; icon: ComponentType<{ className?: string }>; value: string; onChange: (v: string) => void; type?: string; required?: boolean; minLength?: number }) {
  return <label><span className="mb-2 block text-xs font-bold text-slate-600">{label}</span><span className="relative block"><Icon className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" /><input type={type} required={required} minLength={minLength} value={value} onChange={e => onChange(e.target.value)} className="focus-ring w-full rounded-xl border border-slate-200 bg-white py-3 pl-10 pr-3 text-sm font-medium text-slate-800" /></span></label>
}
